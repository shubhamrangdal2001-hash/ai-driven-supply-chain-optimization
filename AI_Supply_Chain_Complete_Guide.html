from statsmodels.tsa.statespace.sarimax import SARIMAX
from statsmodels.tsa.stattools import adfuller
import pandas as pd, numpy as np

# Use single top-selling product for ARIMA
top_product = (
    demand_grouped.groupby('Product_Code')['Order_Demand'].sum()
    .idxmax()
)
ts = (
    demand_grouped[demand_grouped['Product_Code'] == top_product]
    .set_index('Date')['Order_Demand']
    .resample('W').sum()
    .dropna()
)

# Test stationarity
adf_result = adfuller(ts)
print(f"ADF p-value: {adf_result[1]:.4f}")
if adf_result[1] > 0.05:
    ts_diff = ts.diff().dropna()
    print("Applied first-order differencing")
else:
    ts_diff = ts

# Train/test split (80/20)
split = int(len(ts) * 0.8)
train, test = ts[:split], ts[split:]

# Fit SARIMA(1,1,1)(1,1,1,52)
model = SARIMAX(train, order=(1,1,1), seasonal_order=(1,1,1,52),
                enforce_stationarity=False)
sarima_fit = model.fit(disp=False)

# Forecast
forecast = sarima_fit.forecast(steps=len(test))
mae_arima  = np.mean(np.abs(test.values - forecast.values))
mape_arima = np.mean(np.abs((test.values - forecast.values) / test.values)) * 100
rmse_arima = np.sqrt(np.mean((test.values - forecast.values)**2))
print(f"SARIMA  MAE={mae_arima:.1f}  RMSE={rmse_arima:.1f}  MAPE={mape_arima:.1f}%")

# Prepare features
df_feat = demand_grouped.copy()
df_feat['month']      = df_feat['Date'].dt.month
df_feat['quarter']    = df_feat['Date'].dt.quarter
df_feat['year']       = df_feat['Date'].dt.year

le_wh = LabelEncoder()

df_feat['product_enc']   = le_pr.fit_transform(df_feat['Product_Code'])

FEATURES = ['warehouse_enc', 'product_enc', 'month', 'quarter',
            'year', 'dayofyear', 'week',
            'lag_7', 'lag_14', 'lag_28', 'rolling_mean_7', 'rolling_std_7']
TARGET = 'Order_Demand'

df_clean = df_feat[FEATURES + [TARGET, 'Date']].dropna()
cutoff = pd.Timestamp('2016-01-01')
X_train = df_clean[df_clean['Date'] < cutoff][FEATURES]
y_train = df_clean[df_clean['Date'] < cutoff][TARGET]
X_test  = df_clean[df_clean['Date'] >= cutoff][FEATURES]
y_test  = df_clean[df_clean['Date'] >= cutoff][TARGET]

# Hyperparameter tuning with Optuna
def objective(trial):
    params = {
        'max_depth':        trial.suggest_int('max_depth', 3, 10),
        'learning_rate':    trial.suggest_float('lr', 0.01, 0.3, log=True),
        'n_estimators':     trial.suggest_int('n_est', 100, 500),
        'subsample':        trial.suggest_float('sub', 0.6, 1.0),
        'colsample_bytree': trial.suggest_float('col', 0.6, 1.0),
    }
    m = xgb.XGBRegressor(**params, random_state=42, n_jobs=-1)
    m.fit(X_train, y_train, eval_set=[(X_test, y_test)],
          early_stopping_rounds=20, verbose=False)
    preds = m.predict(X_test)
    return mean_absolute_error(y_test, preds)

study = optuna.create_study(direction='minimize')
study.optimize(objective, n_trials=50, show_progress_bar=True)

# Train best model
best_xgb = xgb.XGBRegressor(**study.best_params, random_state=42, n_jobs=-1)
best_xgb.fit(X_train, y_train)
preds_xgb = best_xgb.predict(X_test)

rmse_xgb = np.sqrt(mean_squared_error(y_test, preds_xgb))

print(f"XGBoost MAE={mae_xgb:.1f}  RMSE={rmse_xgb:.1f}  MAPE={mape_xgb:.1f}%")
phase2c_lstm_model.py 
import torch
import torch.nn as nn
from torch.utils.data import DataLoader, TensorDataset
from sklearn.preprocessing import MinMaxScaler

SEQ_LEN   = 30    # look-back window
PRED_LEN  = 7     # forecast horizon
HIDDEN    = 128
LAYERS    = 2
DROPOUT   = 0.2
EPOCHS    = 50
BATCH     = 64
DEVICE    = torch.device('cuda' if torch.cuda.is_available() else 'cpu')

# ── Prepare sequences ──────────────────────────────────────────────
scaler = MinMaxScaler()
values = scaler.fit_transform(ts.values.reshape(-1,1)).flatten()

def make_sequences(data, seq_len, pred_len):
    X, y = [], []
    for i in range(len(data) - seq_len - pred_len + 1):
        X.append(data[i:i+seq_len])
        y.append(data[i+seq_len:i+seq_len+pred_len])
    return np.array(X), np.array(y)

X_all, y_all = make_sequences(values, SEQ_LEN, PRED_LEN)
split = int(len(X_all) * 0.8)
X_tr = torch.FloatTensor(X_all[:split]).unsqueeze(-1).to(DEVICE)
y_tr = torch.FloatTensor(y_all[:split]).to(DEVICE)
X_te = torch.FloatTensor(X_all[split:]).unsqueeze(-1).to(DEVICE)
y_te = torch.FloatTensor(y_all[split:]).to(DEVICE)

loader = DataLoader(TensorDataset(X_tr, y_tr), batch_size=BATCH, shuffle=True)

# ── LSTM Model ────────────────────────────────────────────────────
class DemandLSTM(nn.Module):
    def __init__(self, input_dim=1, hidden=128, layers=2,
                 dropout=0.2, pred_len=7):
        super().__init__()
        self.lstm = nn.LSTM(input_dim, hidden, layers,
                            dropout=dropout, batch_first=True)
        self.norm = nn.LayerNorm(hidden)
        self.head = nn.Sequential(
            nn.Linear(hidden, 64), nn.ReLU(),
            nn.Dropout(dropout),
            nn.Linear(64, pred_len)
        )

    def forward(self, x):
        out, _ = self.lstm(x)
        out    = self.norm(out[:, -1, :])
        return self.head(out)

model_lstm = DemandLSTM(pred_len=PRED_LEN).to(DEVICE)
optimizer  = torch.optim.Adam(model_lstm.parameters(), lr=1e-3)
scheduler  = torch.optim.lr_scheduler.ReduceLROnPlateau(optimizer, patience=5)
criterion  = nn.HuberLoss()

# ── Training loop ─────────────────────────────────────────────────
print("Training LSTM...")
for epoch in range(1, EPOCHS+1):
    model_lstm.train()
    epoch_loss = 0
    for xb, yb in loader:
        optimizer.zero_grad()
        pred = model_lstm(xb)
        loss = criterion(pred, yb)
        loss.backward()
        nn.utils.clip_grad_norm_(model_lstm.parameters(), 1.0)
        optimizer.step()
        epoch_loss += loss.item()
    scheduler.step(epoch_loss)
    if epoch % 10 == 0:
        print(f"  Epoch {epoch}/{EPOCHS}  Loss={epoch_loss/len(loader):.4f}")

# ── Evaluate ──────────────────────────────────────────────────────
model_lstm.eval()
with torch.no_grad():
    preds_lstm_scaled = model_lstm(X_te).cpu().numpy()
preds_lstm = scaler.inverse_transform(
    preds_lstm_scaled.reshape(-1, 1)
).flatten()
actual_lstm = scaler.inverse_transform(y_te.cpu().numpy().reshape(-1, 1)).flatten()

mae_lstm  = mean_absolute_error(actual_lstm, preds_lstm)
rmse_lstm = np.sqrt(mean_squared_error(actual_lstm, preds_lstm))
mape_lstm = np.mean(np.abs((actual_lstm - preds_lstm) / actual_lstm.clip(1))) * 100
print(f"LSTM    MAE={mae_lstm:.1f}  RMSE={rmse_lstm:.1f}  MAPE={mape_lstm:.1f}%")

# Save best model
torch.save(model_lstm.state_dict(), './src/models/lstm_demand.pt')

phase3_route_optimization.py 
from ortools.constraint_solver import routing_enums_pb2
from ortools.constraint_solver import pywrapcp
import numpy as np, pandas as pd
import folium
from math import radians, cos, sin, asin, sqrt

# ── 1. Prepare location data from Olist ───────────────────────────
geo_agg = (
    geo.groupby('geolocation_zip_code_prefix')
    .agg(lat=('geolocation_lat', 'mean'),
         lng=('geolocation_lng', 'mean'))
    .reset_index()
)

# Merge sellers with geo
sellers_geo = sellers.merge(
    geo_agg, left_on='seller_zip_code_prefix',
    right_on='geolocation_zip_code_prefix', how='left'
).dropna(subset=['lat', 'lng'])

# Sample 15 sellers + 1 depot for demo
np.random.seed(42)
sample_locs = sellers_geo.sample(15)[[ 'seller_id', 'lat', 'lng']].reset_index(drop=True)
depot = pd.DataFrame([{'seller_id': 'DEPOT',
                        'lat': -23.5505, 'lng': -46.6333}])  # São Paulo
locations = pd.concat([depot, sample_locs], ignore_index=True)

# ── 2. Haversine distance matrix ─────────────────────────────────
def haversine_km(lat1, lon1, lat2, lon2):
    R = 6371
    lat1, lon1, lat2, lon2 = map(radians, [lat1, lon1, lat2, lon2])
    dlat = lat2 - lat1; dlon = lon2 - lon1
    a = sin(dlat/2)**2 + cos(lat1)*cos(lat2)*sin(dlon/2)**2
    return 2*R*asin(sqrt(a))

n = len(locations)
dist_matrix = np.zeros((n, n), dtype=int)
for i in range(n):
    for j in range(n):
        km = haversine_km(locations.lat[i], locations.lng[i],
                          locations.lat[j], locations.lng[j])
        dist_matrix[i][j] = int(km * 1000)  # metres

# ── 3. Build VRP data model ───────────────────────────────────────
NUM_VEHICLES    = 3
VEHICLE_CAPACITY = 15000   # kg
demands = [0] + [np.random.randint(500, 3000) for _ in range(n-1)]

data = {
    'distance_matrix': dist_matrix.tolist(),
    'demands': demands,
    'vehicle_capacities': [VEHICLE_CAPACITY] * NUM_VEHICLES,
    'num_vehicles': NUM_VEHICLES,
    'depot': 0
}

# ── 4. Solve with OR-Tools ─────────────────────────────────────────
manager  = pywrapcp.RoutingIndexManager(len(data['distance_matrix']),
                                        data['num_vehicles'], data['depot'])
routing  = pywrapcp.RoutingModel(manager)

def dist_callback(from_idx, to_idx):
    f = manager.IndexToNode(from_idx)
    t = manager.IndexToNode(to_idx)
    return data['distance_matrix'][f][t]

transit_cb = routing.RegisterTransitCallback(dist_callback)
routing.SetArcCostEvaluatorOfAllVehicles(transit_cb)

def demand_callback(from_idx):
    node = manager.IndexToNode(from_idx)
    return data['demands'][node]

demand_cb = routing.RegisterUnaryTransitCallback(demand_callback)
routing.AddDimensionWithVehicleCapacity(
    demand_cb, 0, data['vehicle_capacities'], True, 'Capacity'
)

search_params = pywrapcp.DefaultRoutingSearchParameters()
search_params.first_solution_strategy = (
    routing_enums_pb2.FirstSolutionStrategy.PATH_CHEAPEST_ARC
)
search_params.local_search_metaheuristic = (
    routing_enums_pb2.LocalSearchMetaheuristic.GUIDED_LOCAL_SEARCH
)
search_params.time_limit.seconds = 30

solution = routing.SolveWithParameters(search_params)

# ── 5. Extract & display routes ───────────────────────────────────
COLORS = ['red', 'blue', 'green']
route_plan = {}

if solution:
    total_dist = 0
    for v in range(data['num_vehicles']):
        idx = routing.Start(v); route = []; load = 0
        while not routing.IsEnd(idx):
            node = manager.IndexToNode(idx)
            route.append(node); load += data['demands'][node]
            idx = solution.Value(routing.NextVar(idx))
        route.append(0)
        route_dist = sum(
            dist_matrix[route[i]][route[i+1]] for i in range(len(route)-1)
        ) / 1000
        total_dist += route_dist
        route_plan[v] = {'nodes': route, 'load': load, 'km': route_dist}
        print(f"Vehicle {v+1}: {route}  load={load}kg  dist={route_dist:.1f}km")
    print(f"Total distance: {total_dist:.1f} km")

# ── 6. Folium map ─────────────────────────────────────────────────
m = folium.Map(location=[locations.lat.mean(), locations.lng.mean()], zoom_start=7)

for v, rdata in route_plan.items():
    coords = [(locations.lat[n], locations.lng[n]) for n in rdata['nodes']]
    folium.PolyLine(coords, color=COLORS[v], weight=3,
                    tooltip=f"Vehicle {v+1} ({rdata['km']:.0f} km)").add_to(m)

for i, row in locations.iterrows():
    color = 'black' if i == 0 else 'gray'
    icon  = 'home'  if i == 0 else 'circle'
    folium.Marker([row.lat, row.lng],
                  popup=row.seller_id,
                  icon=folium.Icon(color=color, icon=icon)).add_to(m)

m.save('./reports/optimized_routes.html')
print("Route map saved.")

import numpy as np, pandas as pd
from scipy.stats import norm
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import classification_report

SERVICE_LEVEL = 0.95          # 95% in-stock probability
Z_SCORE       = norm.ppf(SERVICE_LEVEL)  # = 1.645

# ── 1. Compute demand statistics per SKU ─────────────────────────
sku_stats = (
    demand_grouped.groupby('Product_Code')['Order_Demand']
    .agg(
        avg_daily_demand='mean',
        std_daily_demand='std',
        total_demand='sum',
        days_active='count'
    ).reset_index()
)
sku_stats['cv'] = sku_stats['std_daily_demand'] / sku_stats['avg_daily_demand'].clip(1)

# ── 2. EOQ — Economic Order Quantity ──────────────────────────────
ORDERING_COST   = 50    # $ per order
HOLDING_COST_PCT = 0.2  # 20% of unit cost per year
AVG_UNIT_COST   = 10    # $10 per unit
HOLDING_COST    = HOLDING_COST_PCT * AVG_UNIT_COST

sku_stats['annual_demand'] = sku_stats['avg_daily_demand'] * 365
sku_stats['EOQ'] = np.sqrt(
    (2 * sku_stats['annual_demand'] * ORDERING_COST) / HOLDING_COST
).round(0)

# ── 3. Safety Stock & Reorder Point ─────────────────────────────
LEAD_TIME_DAYS      = 7
LEAD_TIME_STD_DAYS  = 2

sku_stats['demand_during_lead'] = sku_stats['avg_daily_demand'] * LEAD_TIME_DAYS
sku_stats['safety_stock'] = (
    Z_SCORE * np.sqrt(
        LEAD_TIME_DAYS * sku_stats['std_daily_demand']**2
        + (sku_stats['avg_daily_demand'] * LEAD_TIME_STD_DAYS)**2
    )
).round(0)

sku_stats['reorder_point'] = (
    sku_stats['demand_during_lead'] + sku_stats['safety_stock']
).round(0)

# ── 4. ABC Classification ────────────────────────────────────────
sku_stats['revenue_pct'] = sku_stats['total_demand'] / sku_stats['total_demand'].sum()
sku_stats = sku_stats.sort_values('revenue_pct', ascending=False)
sku_stats['cumulative_pct'] = sku_stats['revenue_pct'].cumsum()

def abc_class(cum):
    if   cum <= 0.80: return 'A'
    elif cum <= 0.95: return 'B'
    else:             return 'C'

sku_stats['ABC'] = sku_stats['cumulative_pct'].apply(abc_class)
print(sku_stats['ABC'].value_counts())

# ── 5. Stockout Risk Classifier ───────────────────────────────────
# Label: stockout if order demand > safety_stock threshold
df_sc = df[[ 'Order Item Quantity', 'Days for shipment (scheduled)',
             'Days for shipping (real)', 'order_month',
             'order_quarter', 'Shipping Mode', 'Category Name']].dropna()
df_sc['stockout_risk'] = (
    df_sc['Days for shipping (real)'] > df_sc['Days for shipment (scheduled)'] * 1.5
).astype(int)

cat_cols = ['Shipping Mode', 'Category Name']
df_enc = pd.get_dummies(df_sc, columns=cat_cols, drop_first=True)
feat_cols = [c for c in df_enc.columns if c != 'stockout_risk']

X = df_enc[feat_cols].values
y = df_enc['stockout_risk'].values
split = int(len(X) * 0.8)

clf = RandomForestClassifier(n_estimators=200, max_depth=8,
                              class_weight='balanced', random_state=42, n_jobs=-1)
clf.fit(X[:split], y[:split])
print(classification_report(y[split:], clf.predict(X[split:])))

# Save inventory summary
sku_stats.to_csv('./reports/inventory_plan.csv', index=False)
print("Inventory plan saved.")

from fastapi import FastAPI
from pydantic import BaseModel
import numpy as np, torch, joblib, uvicorn
from typing import List

app = FastAPI(title="Supply Chain AI API", version="1.0")

# Load models at startup

lstm_model.eval()
sku_plan   = pd.read_csv("./reports/inventory_plan.csv")

class DemandRequest(BaseModel):
    product_code: str
    warehouse:    str
    horizon_days: int = 7
    recent_demand: List[float]   # last 30 daily values

class RouteRequest(BaseModel):
    depot_lat:  float; depot_lng:  float
    stops:      List[dict]       # [{"lat":..,"lng":..,"demand":..}]
    n_vehicles: int = 3

@app.get("/")
def root():
    return {"status": "Supply Chain AI API running"}

@app.post("/predict/demand")
def predict_demand(req: DemandRequest):
    seq = np.array(req.recent_demand[-30:], dtype=np.float32)
    seq_scaled = scaler.transform(seq.reshape(-1,1)).flatten()
    tensor = torch.FloatTensor(seq_scaled).unsqueeze(0).unsqueeze(-1)
    with torch.no_grad():
        pred = lstm_model(tensor).numpy().flatten()
    forecast = scaler.inverse_transform(pred.reshape(-1,1)).flatten().tolist()
    return {
        "product_code":   req.product_code,
        "warehouse":      req.warehouse,
        "forecast_7days": forecast,
        "total_forecast": round(sum(forecast), 1)
    }

@app.get("/inventory/{product_code}")

    row = sku_plan[sku_plan['Product_Code'] == product_code]

    return row.iloc[0][[
        'Product_Code', 'EOQ', 'safety_stock',
        'reorder_point', 'ABC'
    ]].to_dict()

@app.get("/health")
